% Link colours. Muted enough to stay readable when the book is printed in
% greyscale, saturated enough on screen to read as "this is clickable".
% Internal references and citations share one dark blue; external URLs get a
% warmer indigo so a reader can tell at a glance whether a link leaves the book.
\definecolor{snnLink}{rgb}{0.05, 0.24, 0.52}
\definecolor{snnUrl}{rgb}{0.27, 0.20, 0.55}
\hypersetup{
  colorlinks,
  linkcolor={snnLink},
  citecolor={snnLink},
  urlcolor={snnUrl},
  linktoc=all
}

% Style quotes
\definecolor{darkblue}{rgb}{0.0, 0.0, 0.55}
\definecolor{quoteshade}{rgb}{0.95, 0.95, 1}
\renewenvironment{quote}{%
  \def\FrameCommand{%
    \hspace{1pt}%
    {\color{darkblue}\vrule width 2pt}%
    {\color{quoteshade}\vrule width 4pt}%
    \colorbox{quoteshade}%
  }%
  \MakeFramed {\advance\hsize-\width \FrameRestore}%
  \noindent\hspace{-8pt}% disable indenting first paragraph
  \begin{adjustwidth}{0pt}{0pt}% adjust as needed
  \vspace{2pt}\vspace{2pt}%
}
{%
  \vspace{2pt}\end{adjustwidth}\endMakeFramed%
}

% Style code listings: line numbers + smaller font.
% MyST emits code blocks as {verbatim}; fancyvrb lets us recustom that
% environment in place so every code block across the book picks up the style.
\usepackage{fancyvrb}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{%
  fontsize=\small,%
  numbers=left,%
  numbersep=8pt,%
  xleftmargin=2.5em,%
  frame=leftline,%
  framesep=6pt,%
}

% Relabel the sectioning hierarchy for the print book:
%   top-level \chapter  -> "Topic"   (was "Chapter")
%   next-level \section -> "Chapter"
\usepackage{titlesec}
\renewcommand{\chaptername}{Topic}
\titleformat{\section}
  {\normalfont\Large\bfseries}   % heading font
  {Chapter \thesection}          % label printed before the title
  {1em}                          % gap between label and title
  {}

% Front matter (acknowledgements, preface): roman PAGE numbers, and chapters
% numbered with uppercase roman shown inline as "I. Acknowledgements".
% \frontmatter gives roman pages but disables chapter numbers, so \@mainmattertrue
% re-enables them. secnumdepth 0 numbers chapters only, leaving front-matter
% sections unnumbered. The book body restores the default "Topic N" style and
% arabic numbering at the front/main boundary (see topics/0_why_snns.md).
\frontmatter
\makeatletter
\@mainmattertrue
\renewcommand{\thechapter}{\Roman{chapter}}
\renewcommand\@makechapterhead[1]{%
  \vspace*{50\p@}%
  {\parindent\z@ \raggedright \normalfont
    \huge\bfseries \thechapter.\space #1\par\nobreak
    \vskip 40\p@}}
\makeatother
\setcounter{secnumdepth}{0}
\maketitle
